cmake_minimum_required(VERSION 3.20)
project(atlas_sample_benchmark C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ATLAS_ORIGINAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/original" CACHE PATH "Read-only original source directory")
option(ATLAS_SANITIZE "Check the benchmark with address and undefined sanitizers" OFF)
if(ATLAS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
file(GLOB ORIGINAL_SOURCES "${ATLAS_ORIGINAL_ROOT}/source/*.c")
list(FILTER ORIGINAL_SOURCES EXCLUDE REGEX "/main.c$")
# Observe matched IDs only in the instrumented executable. The replacement adds
# one callback after the real predicate, without editing the engine source.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine-source/engine.c" OBSERVED_ENGINE)
set(MATCH_SITE "if(!matches(r,f))return;s->total++;")
string(FIND "${OBSERVED_ENGINE}" "${MATCH_SITE}" MATCH_POSITION)
if(MATCH_POSITION EQUAL -1)
  message(FATAL_ERROR "Engine visitor changed; review the benchmark observer")
endif()
string(REPLACE "${MATCH_SITE}" "if(!matches(r,f))return;bench_visit_id(r->id);s->total++;" OBSERVED_ENGINE "${OBSERVED_ENGINE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine-observed.c" "#include <stdint.h>\nextern void bench_visit_id(uint32_t id);\n${OBSERVED_ENGINE}")
foreach(mode IN ITEMS timing memory)
  add_library(original_${mode} STATIC ${ORIGINAL_SOURCES})
  target_include_directories(original_${mode} PUBLIC "${ATLAS_ORIGINAL_ROOT}/include")
  if(mode STREQUAL "memory")
    add_library(engine_${mode} STATIC "${CMAKE_CURRENT_BINARY_DIR}/engine-observed.c")
  else()
    add_library(engine_${mode} STATIC ../engine-source/engine.c)
  endif()
  target_include_directories(engine_${mode} PUBLIC ../engine-source)
  add_executable(bench_${mode} src/compare.c src/alloc.c)
  target_link_libraries(bench_${mode} PRIVATE original_${mode} engine_${mode} m)
  target_include_directories(bench_${mode} PRIVATE src)
  target_compile_options(bench_${mode} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
# Instrument allocations in a separate executable. Timing uses the unmodified allocator.
foreach(target IN ITEMS original_memory engine_memory)
  target_compile_definitions(${target} PRIVATE ATLAS_TRACK_ALLOCATIONS=1)
  target_compile_options(${target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/src/alloc.h")
endforeach()
target_compile_definitions(bench_memory PRIVATE ATLAS_MEMORY_RUN=1)
enable_testing()
foreach(mode IN ITEMS timing memory)
  add_test(NAME sample_${mode} COMMAND bench_${mode}
    "${CMAKE_CURRENT_SOURCE_DIR}/sample/sinistros-1000.csv"
    "${CMAKE_CURRENT_SOURCE_DIR}/sample/records-1000.bin")
endforeach()
